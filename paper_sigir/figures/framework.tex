% Helper command for circled numbers - 必须在 begin{document} 前定义
\newcommand*\circled[1]{%
  \tikz[baseline=(char.base)]{
    \node[shape=circle, draw, inner sep=1.5pt, font=\footnotesize\bfseries, 
          line width=0.9pt, minimum size=0.5cm] (char) {#1};
  }%
}
% 白色版本 - 用于深色背景
\newcommand*\circledw[1]{%
  \tikz[baseline=(char.base)]{
    \node[shape=circle, draw=white, inner sep=1.5pt, font=\footnotesize\bfseries, 
          text=white, line width=0.9pt, minimum size=0.5cm] (char) {#1};
  }%
}
